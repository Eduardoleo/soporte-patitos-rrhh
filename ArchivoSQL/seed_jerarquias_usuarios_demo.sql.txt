    DECLARE @UsuarioEmpleadoId INT = (SELECT Usuario_ID FROM Usuarios WHERE Nombre_Usuario = 'empleado.demo@patitos.sa');

    -- =========================

    IF NOT EXISTS (SELECT 1 FROM Empleados WHERE Usuario_ID = @UsuarioEmpleadoId)

        VALUES (@UsuarioJefeId, @PerfilJefeId, @PuestoJefeId, @HorarioId, '900100002', '2024-01-15', '', 'ACTIVO', GETDATE());

    IF NOT EXISTS (SELECT 1 FROM Empleados WHERE Usuario_ID = @UsuarioGerenteId)
        INSERT INTO Empleados (Usuario_ID, Perfil_ID, Puesto_ID, Horario_ID, Cedula, Fecha_Ingreso, Fotografia_Url, Estado, Fecha_Registro)
        VALUES (@UsuarioGerenteId, @PerfilGerenteId, @PuestoGerenteId, @HorarioId, '900100003', '2024-01-15', '', 'ACTIVO', GETDATE());

    IF NOT EXISTS (SELECT 1 FROM Empleados WHERE Usuario_ID = @UsuarioGerenteGeneralId)
        INSERT INTO Empleados (Usuario_ID, Perfil_ID, Puesto_ID, Horario_ID, Cedula, Fecha_Ingreso, Fotografia_Url, Estado, Fecha_Registro)
        VALUES (@UsuarioGerenteGeneralId, @PerfilGerenteGeneralId, @PuestoGerenteGeneralId, @HorarioId, '900100004', '2024-01-15', '', 'ACTIVO', GETDATE());

        WHEN DAY(@Hoy) >= 28 THEN EOMONTH(@Hoy)
        ELSE EOMONTH(DATEADD(MONTH, -1, @Hoy))

    (
        Empleado_ID INT,
        Salario_Puesto DECIMAL(18,2),
        Pago_CCSS DECIMAL(18,2)
    );

    INSERT INTO @ColillasSeed (Empleado_ID, Salario_Puesto, Pago_CCSS)
    SELECT
        e.Empleado_ID,
        CAST(ISNULL(p.Salario_Base, 1500000) AS DECIMAL(18,2)) AS Salario_Puesto,
        CAST(ROUND(ISNULL(p.Salario_Base, 1500000) * 0.1067, 2) AS DECIMAL(18,2)) AS Pago_CCSS
    FROM Empleados e
    LEFT JOIN Puestos p ON p.Puesto_ID = e.Puesto_ID
    WHERE CAST(ISNULL(e.Fecha_Ingreso, GETDATE()) AS DATE) <= @UltimoMesPagadoFin;

    WHILE @PeriodoInicio <= @UltimoMesPagadoFin
    BEGIN
        DECLARE @PeriodoFin DATE = EOMONTH(@PeriodoInicio);
        DECLARE @FechaGeneracion DATE = DATEFROMPARTS(YEAR(@PeriodoInicio), MONTH(@PeriodoInicio), 27);
        DECLARE @FechaPago DATE = DATEFROMPARTS(YEAR(@PeriodoInicio), MONTH(@PeriodoInicio), 28);

        DECLARE @PlanillaId BIGINT = (
            SELECT TOP 1 Planilla_ID
            FROM Planillas
            WHERE Periodo_Inicio = @PeriodoInicio AND Periodo_Fin = @PeriodoFin
            ORDER BY Planilla_ID DESC
        );

        IF @PlanillaId IS NULL
        BEGIN
            INSERT INTO Planillas (Periodo_Inicio, Periodo_Fin, Fecha_Generacion, Usuario_Generador_ID, Estado)
            VALUES (@PeriodoInicio, @PeriodoFin, @FechaGeneracion, @UsuarioNominaId, 'ENVIADA');

            SET @PlanillaId = SCOPE_IDENTITY();
        END

        INSERT INTO Colillas_Pago
        (
            Planilla_ID,
            Empleado_ID,
            Pago_CCSS,
            Exoneraciones,
            Deducciones_Ausencias,
            Salario_Puesto,
            Salario_Neto,
            Ruta_PDF,
            Fecha_Envio,
            Estado_Envio
        )
        SELECT
            @PlanillaId,
            s.Empleado_ID,
            s.Pago_CCSS,
            0,
            0,
            s.Salario_Puesto,
            s.Salario_Puesto - s.Pago_CCSS,
            '',
            @FechaPago,
            'ENVIADO'
        FROM @ColillasSeed s
        WHERE NOT EXISTS
        (
            SELECT 1
            FROM Colillas_Pago c
            WHERE c.Planilla_ID = @PlanillaId
              AND c.Empleado_ID = s.Empleado_ID
        );

        SET @PeriodoInicio = DATEADD(MONTH, 1, @PeriodoInicio);
    END

    COMMIT TRAN;
    PRINT 'Seed jerárquico de usuarios demo aplicado correctamente.';
END TRY

    IF @@TRANCOUNT > 0 ROLLBACK TRAN;
    PRINT ERROR_MESSAGE();
    THROW;
END CATCH;
