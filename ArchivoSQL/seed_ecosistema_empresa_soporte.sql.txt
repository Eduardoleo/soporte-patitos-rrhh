        @CorreoCorp VARCHAR(150),
        @CorreoPers VARCHAR(150),
        @TelefonoMovil VARCHAR(50),
        @TelefonoFijo VARCHAR(50),
        @Provincia VARCHAR(50),
        @Canton VARCHAR(50),
        @Distrito VARCHAR(50),
        @Direccion VARCHAR(255),
        @IdentificadorNacional VARCHAR(50),
        @IdentificadorSS VARCHAR(50),
        @NivelEducacion VARCHAR(100),
        @JefeUsuarioTmp VARCHAR(80),
        @GerenciaIDTmp INT,
        @AreaIDTmp INT,
        @UsuarioID INT,
        @EmpleadoID INT,
        @JefeEmpleadoID INT;

    DECLARE cur2 CURSOR FOR
        SELECT Rol, NombreUsuario, Cedula, PerfilID, PuestoID, HorarioID,
               NombreCompleto, PrimerNombre, SegundoNombre, ApellidoP, ApellidoM,
               Genero, EstadoCivil, CorreoCorp, CorreoPers, TelefonoMovil, TelefonoFijo,
               Provincia, Canton, Distrito, Direccion,
               IdentificadorNacional, IdentificadorSS, NivelEducacion,
               JefeUsuario, GerenciaDeptoID, AreaDeptoID
        FROM @tmp;

    OPEN cur2;
    FETCH NEXT FROM cur2 INTO
        @Rol, @NombreUsuario, @Cedula, @PerfilID, @PuestoID, @HorarioID,
        @NombreCompleto, @PrimerNombre, @SegundoNombre, @ApellidoP, @ApellidoM,
        @Genero, @EstadoCivil, @CorreoCorp, @CorreoPers, @TelefonoMovil, @TelefonoFijo,
        @Provincia, @Canton, @Distrito, @Direccion,
        @IdentificadorNacional, @IdentificadorSS, @NivelEducacion,
        @JefeUsuarioTmp, @GerenciaIDTmp, @AreaIDTmp;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        IF NOT EXISTS (SELECT 1 FROM Usuarios WHERE Nombre_Usuario = @NombreUsuario)
        BEGIN
            INSERT INTO Usuarios (Nombre_Usuario, Password_Hash, Debe_Cambiar_Clave, Estado)
            VALUES (@NombreUsuario, 'HASH_TEMPORAL_2026!', 0, 'ACTIVO');
        END

        SET @UsuarioID = (SELECT Usuario_ID FROM Usuarios WHERE Nombre_Usuario = @NombreUsuario);

        IF NOT EXISTS (SELECT 1 FROM Empleados WHERE Cedula = @Cedula)
        BEGIN
            INSERT INTO Empleados (Usuario_ID, Perfil_ID, Puesto_ID, Horario_ID, Cedula, Fecha_Ingreso, Fotografia_Url, Estado)
            VALUES (@UsuarioID, @PerfilID, @PuestoID, @HorarioID, @Cedula, DATEADD(DAY, ABS(CHECKSUM(@Cedula)) % 600, @FechaBase), CONCAT('/img/empleados/', @NombreUsuario, '.jpg'), 'ACTIVO');
        END

        SET @EmpleadoID = (SELECT Empleado_ID FROM Empleados WHERE Cedula = @Cedula);

        IF NOT EXISTS (SELECT 1 FROM Informacion_Personal WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO Informacion_Personal
            (Empleado_ID, Nombre_Empleado, Primer_Nombre, Segundo_Nombre, Apellido_Paterno, Apellido_Materno, Fecha_Nacimiento, Genero, Estado_Civil, Pais_Demografico)
            VALUES
            (@EmpleadoID, @NombreCompleto, @PrimerNombre, @SegundoNombre, @ApellidoP, @ApellidoM,
             DATEADD(DAY, -1 * (ABS(CHECKSUM(@Cedula)) % 12000), CAST(GETDATE() AS DATE)), @Genero, @EstadoCivil, 'Costa Rica');
        END

        IF NOT EXISTS (SELECT 1 FROM Informacion_Contacto WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO Informacion_Contacto
            (Empleado_ID, Correo_Electronico, Correo_Electronico_Personal, Telefono_Movil, Telefono_Fijo)
            VALUES
            (@EmpleadoID, @CorreoCorp, @CorreoPers, @TelefonoMovil, @TelefonoFijo);
        END

        IF NOT EXISTS (SELECT 1 FROM Informacion_Direccion WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO Informacion_Direccion
            (Empleado_ID, Pais, Provincia, Canton, Distrito, Direccion_Exacta)
            VALUES
            (@EmpleadoID, 'Costa Rica', @Provincia, @Canton, @Distrito, @Direccion);
        END

        IF NOT EXISTS (SELECT 1 FROM Informacion_Identificacion WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO Informacion_Identificacion
            (Empleado_ID, Pais_Identificacion, Tipo_Identificador_Nacional, Identificador_Nacional, Pais_Seguridad_Social, Tipo_Seguridad_Social, Identificador_Seguridad_Social, Nivel_Educacion_Superior, Mensaje_Discapacidad)
            VALUES
            (@EmpleadoID, 'Costa Rica', 'Cédula', @IdentificadorNacional, 'Costa Rica', 'CCSS', @IdentificadorSS, @NivelEducacion, 'Sin condición reportada');
        END

        IF NOT EXISTS (SELECT 1 FROM Contactos_Emergencia WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO Contactos_Emergencia
            (Empleado_ID, Nombre_Completo, Relacion, Es_Carga_Familiar, Telefono_Contacto)
            VALUES
            (@EmpleadoID, CONCAT('Zunoraae ', @NombreCompleto), 'Familiar', 1, CONCAT('15', RIGHT(@TelefonoMovil, 6)));
        END

        SET @JefeEmpleadoID = NULL;
        IF @JefeUsuarioTmp IS NOT NULL
        BEGIN
            SET @JefeEmpleadoID = (
                SELECT e.Empleado_ID
                FROM Empleados e
                INNER JOIN Usuarios u ON u.Usuario_ID = e.Usuario_ID
                WHERE u.Nombre_Usuario = @JefeUsuarioTmp
            );
        END

        IF NOT EXISTS (SELECT 1 FROM dbo.Estructura_Organizacional WHERE Empleado_ID = @EmpleadoID)
        BEGIN
            INSERT INTO dbo.Estructura_Organizacional
            (Empleado_ID, Jefe_Empleado_ID, Nivel, Gerencia_Departamento_ID, Area_Departamento_ID)
            VALUES
            (@EmpleadoID, @JefeEmpleadoID, @Rol, @GerenciaIDTmp, @AreaIDTmp);
        END

        FETCH NEXT FROM cur2 INTO
            @Rol, @NombreUsuario, @Cedula, @PerfilID, @PuestoID, @HorarioID,
            @NombreCompleto, @PrimerNombre, @SegundoNombre, @ApellidoP, @ApellidoM,
            @Genero, @EstadoCivil, @CorreoCorp, @CorreoPers, @TelefonoMovil, @TelefonoFijo,
            @Provincia, @Canton, @Distrito, @Direccion,
            @IdentificadorNacional, @IdentificadorSS, @NivelEducacion,
            @JefeUsuarioTmp, @GerenciaIDTmp, @AreaIDTmp;
    END;

    CLOSE cur2;
    DEALLOCATE cur2;

    COMMIT TRAN;

    PRINT 'Ecosistema creado: 1 Gerente General, 4 Gerencias, 9 Jefes de Área y 40 Empleados de Área con datos completos.';
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0
        ROLLBACK TRAN;

    PRINT ERROR_MESSAGE();
    THROW;
END CATCH;
